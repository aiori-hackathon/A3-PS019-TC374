Algorithm: Two-Tier Evidence Discovery

Core Engine 
Algorithm PersonalDataDiscoveryAndEvidenceInventory(D, R, G)
// Input: Datasets D, Detection Rules R, Governance Registry G
// Output: Technical Privacy Evidence Inventory + Precision/Recall Metrics

Initialize Inventory ← []

for each dataset d in D do
  for each column c in d.columns do
    samples ← SampleNonEmptyValues(c, limit=100)
    best_class ← Non_PII,  max_confidence ← 0.0

    for each class C in R.classes do
      // ── TIER 1: Semantic Header Analysis ──
      header_score ← RegexMatch(c.name, C.header_patterns) ? 1.0 : 0.0

      // ── TIER 2: Statistical Sample Validation ──
      sample_ratio ← CountMatches(samples, C.value_patterns) / |samples|
      value_score  ← min(1.0, sample_ratio / C.min_sample_ratio)

      // ── Combined Confidence with Cross-validation Bonus ──
      confidence ← (header_score × Wh) + (value_score × Wv)
      if header_score > 0 and matching_samples > 0 then
        confidence ← min(1.0, confidence + 0.10)

      if confidence ≥ T and confidence > max_confidence then
        best_class ← C,  max_confidence ← confidence

    // ── Governance Metadata Linking + Gap Analysis ──
    metadata ← LookupGovernance(G, d.name, c.name)
    gaps ← []
    if best_class ≠ Non_PII then
      if metadata.purpose is NULL then gaps ← gaps ∪ {GAP_MISSING_PURPOSE}
      if metadata.retention is NULL then gaps ← gaps ∪ {GAP_MISSING_RETENTION}

    Inventory ← Inventory ∪ {
      field: c.name, class: best_class, confidence: max_confidence,
      purpose: metadata.purpose, retention: metadata.retention,
      gaps: gaps, legal_disclaimer: "Technical Evidence Only — Not Legal Advice"
    }

return Inventory
